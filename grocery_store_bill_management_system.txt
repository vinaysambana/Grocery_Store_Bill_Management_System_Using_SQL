-- ============================================================================================================================
-- PART 1: DATABASE & TABLE CREATION (7 tables; no window functions)-----
-- =============================================================================================================================
CREATE TABLE Categories (
 category_id INT AUTO_INCREMENT PRIMARY KEY,
 category_name VARCHAR(80) NOT NULL UNIQUE,
 description VARCHAR(255)
);

CREATE TABLE Suppliers (
 supplier_id INT AUTO_INCREMENT PRIMARY KEY,
 supplier_name VARCHAR(120) NOT NULL, contact_phone VARCHAR(15),
 email VARCHAR(120), city VARCHAR(100)
);

CREATE TABLE Products (
 product_id INT AUTO_INCREMENT PRIMARY KEY, product_name VARCHAR(120) NOT NULL,
 category_id INT NOT NULL, supplier_id INT NOT NULL,
 unit_price DECIMAL(10,2) NOT NULL, cost_price DECIMAL(10,2) NOT NULL,
 stock_quantity INT UNSIGNED NOT NULL DEFAULT 0,
 reorder_level INT UNSIGNED NOT NULL DEFAULT 10,
 expiry_date DATE,
 CHECK (unit_price >= 0 AND cost_price >= 0),
 FOREIGN KEY (category_id) REFERENCES Categories(category_id),
 FOREIGN KEY (supplier_id) REFERENCES Suppliers(supplier_id)
);

CREATE TABLE Customers (
 customer_id INT AUTO_INCREMENT PRIMARY KEY, customer_name VARCHAR(120) NOT NULL,
 phone VARCHAR(15) UNIQUE, email VARCHAR(120), city VARCHAR(100),
 loyalty_points INT UNSIGNED NOT NULL DEFAULT 0
);

CREATE TABLE Employees (
 employee_id INT AUTO_INCREMENT PRIMARY KEY, employee_name VARCHAR(120) NOT NULL,
 role VARCHAR(50) NOT NULL, hire_date DATE, salary DECIMAL(10,2)
);

CREATE TABLE Bills (
 bill_id INT AUTO_INCREMENT PRIMARY KEY, customer_id INT, employee_id INT NOT NULL,
 bill_date DATE NOT NULL, bill_time TIME NOT NULL,
 payment_method ENUM('Cash','Card','UPI','Credit') NOT NULL DEFAULT 'Cash',
 payment_status ENUM('Paid','Pending','Overdue') NOT NULL DEFAULT 'Paid',
 subtotal DECIMAL(10,2) NOT NULL DEFAULT 0, discount_amount DECIMAL(10,2) NOT NULL DEFAULT 0,
 tax_amount DECIMAL(10,2) NOT NULL DEFAULT 0, total_amount DECIMAL(10,2) NOT NULL DEFAULT 0,
 CHECK (discount_amount >= 0 AND total_amount >= 0),
 FOREIGN KEY (customer_id) REFERENCES Customers(customer_id),
 FOREIGN KEY (employee_id) REFERENCES Employees(employee_id)
);

CREATE TABLE Bill_Items (
 item_id INT AUTO_INCREMENT PRIMARY KEY, bill_id INT NOT NULL, product_id INT NOT NULL,
 quantity INT UNSIGNED NOT NULL, unit_price DECIMAL(10,2) NOT NULL,
 line_total DECIMAL(10,2) NOT NULL DEFAULT 0,
 CHECK (quantity > 0),
 FOREIGN KEY (bill_id) REFERENCES Bills(bill_id),
 FOREIGN KEY (product_id) REFERENCES Products(product_id)
);

CREATE OR REPLACE VIEW view_bill_overview AS
SELECT b.bill_id,b.bill_date,b.bill_time,COALESCE(c.customer_name,'Walk-in') AS customer_name,
 e.employee_name AS cashier,b.payment_method,b.payment_status,
 b.subtotal,b.discount_amount,b.tax_amount,b.total_amount
FROM Bills b LEFT JOIN Customers c ON b.customer_id=c.customer_id
JOIN Employees e ON b.employee_id=e.employee_id;

CREATE OR REPLACE VIEW view_sales_details AS
SELECT bi.item_id,b.bill_id,b.bill_date,p.product_name,cat.category_name,
 bi.quantity,bi.unit_price,bi.line_total
FROM Bill_Items bi JOIN Bills b ON bi.bill_id=b.bill_id
JOIN Products p ON bi.product_id=p.product_id
JOIN Categories cat ON p.category_id=cat.category_id;

CREATE OR REPLACE VIEW view_product_stock_status AS
SELECT p.product_id,p.product_name,cat.category_name,s.supplier_name,
 p.stock_quantity,p.reorder_level,
 CASE WHEN p.stock_quantity<p.reorder_level THEN 'Low Stock' ELSE 'In Stock' END AS stock_status,
 p.expiry_date,
 CASE WHEN p.expiry_date IS NULL THEN 'No Expiry'
      WHEN p.expiry_date<CURDATE() THEN 'Expired'
      WHEN p.expiry_date<=DATE_ADD(CURDATE(),INTERVAL 7 DAY) THEN 'Expiring Soon'
      ELSE 'OK' END AS expiry_status
FROM Products p JOIN Categories cat ON p.category_id=cat.category_id
JOIN Suppliers s ON p.supplier_id=s.supplier_id;

CREATE OR REPLACE VIEW view_category_sales_summary AS
SELECT cat.category_id,cat.category_name,
 COALESCE(SUM(bi.quantity),0) AS units_sold,
 COALESCE(SUM(bi.line_total),0) AS revenue
FROM Categories cat LEFT JOIN Products p ON cat.category_id=p.category_id
LEFT JOIN Bill_Items bi ON p.product_id=bi.product_id
GROUP BY cat.category_id,cat.category_name;

DROP PROCEDURE IF EXISTS sp_bills_by_customer;
DELIMITER $$
CREATE PROCEDURE sp_bills_by_customer(IN p_customer_id INT)
BEGIN
 SELECT bill_id,bill_date,bill_time,payment_method,payment_status,subtotal,discount_amount,tax_amount,total_amount
 FROM Bills WHERE customer_id=p_customer_id ORDER BY bill_date DESC,bill_time DESC;
END $$
DELIMITER ;

DROP PROCEDURE IF EXISTS sp_product_sales_summary;
DELIMITER $$
CREATE PROCEDURE sp_product_sales_summary(IN p_product_id INT)
BEGIN
 SELECT p.product_name,COUNT(DISTINCT bi.bill_id) AS bills_count,
 COALESCE(SUM(bi.quantity),0) AS units_sold,
 COALESCE(SUM(bi.line_total),0) AS revenue,
 p.stock_quantity AS current_stock
 FROM Products p LEFT JOIN Bill_Items bi ON p.product_id=bi.product_id
 WHERE p.product_id=p_product_id GROUP BY p.product_id,p.product_name,p.stock_quantity;
END $$
DELIMITER ;
-- =========================================================================================================================
-- Adds an item using the product's current price, then recalculates the bill totals (5% tax).---
-- ==========================================================================================================================
DROP PROCEDURE IF EXISTS sp_add_bill_item;
DELIMITER $$
CREATE PROCEDURE sp_add_bill_item(IN p_bill_id INT,IN p_product_id INT,IN p_quantity INT)
BEGIN
 DECLARE v_price DECIMAL(10,2) DEFAULT NULL;
 SELECT unit_price INTO v_price FROM Products WHERE product_id=p_product_id;
 IF v_price IS NULL THEN
  SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT='Product not found';
 ELSEIF p_quantity<=0 THEN
  SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT='Quantity must be greater than zero';
 ELSE
  INSERT INTO Bill_Items(bill_id,product_id,quantity,unit_price)
  VALUES(p_bill_id,p_product_id,p_quantity,v_price);
  UPDATE Bills SET subtotal=(SELECT SUM(line_total) FROM Bill_Items WHERE bill_id=p_bill_id)
  WHERE bill_id=p_bill_id;
  UPDATE Bills SET tax_amount=ROUND((subtotal-discount_amount)*0.05,2),
   total_amount=subtotal-discount_amount+ROUND((subtotal-discount_amount)*0.05,2)
  WHERE bill_id=p_bill_id;
 END IF;
END $$
DELIMITER ;

-- PART 6: TRIGGERS---
-- ============================================================================================================================
CREATE TABLE IF NOT EXISTS price_audit (
 audit_id INT AUTO_INCREMENT PRIMARY KEY,product_id INT NOT NULL,
 action_type VARCHAR(10) NOT NULL,old_price DECIMAL(10,2),new_price DECIMAL(10,2),
 changed_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
);

DROP TRIGGER IF EXISTS trg_validate_bill_item_insert;
DELIMITER $$
CREATE TRIGGER trg_validate_bill_item_insert BEFORE INSERT ON Bill_Items
FOR EACH ROW
BEGIN
 DECLARE v_stock INT DEFAULT NULL;
 SELECT stock_quantity INTO v_stock FROM Products WHERE product_id=NEW.product_id;
 IF NEW.quantity<=0 THEN
  SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT='Quantity must be greater than zero';
 ELSEIF v_stock IS NULL OR v_stock<NEW.quantity THEN
  SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT='Insufficient stock for this product';
 END IF;
 SET NEW.line_total=NEW.quantity*NEW.unit_price;
END $$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_reduce_stock_after_sale;
DELIMITER $$
CREATE TRIGGER trg_reduce_stock_after_sale AFTER INSERT ON Bill_Items
FOR EACH ROW
BEGIN
 UPDATE Products SET stock_quantity=stock_quantity-NEW.quantity WHERE product_id=NEW.product_id;
END $$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_audit_product_insert;
DELIMITER $$
CREATE TRIGGER trg_audit_product_insert AFTER INSERT ON Products
FOR EACH ROW
BEGIN
 INSERT INTO price_audit(product_id,action_type,old_price,new_price)
 VALUES(NEW.product_id,'INSERT',NULL,NEW.unit_price);
END $$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_audit_product_price_update;
DELIMITER $$
CREATE TRIGGER trg_audit_product_price_update AFTER UPDATE ON Products
FOR EACH ROW
BEGIN
 IF OLD.unit_price<>NEW.unit_price THEN
  INSERT INTO price_audit(product_id,action_type,old_price,new_price)
  VALUES(NEW.product_id,'UPDATE',OLD.unit_price,NEW.unit_price);
 END IF;
END $$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_audit_product_delete;
DELIMITER $$
CREATE TRIGGER trg_audit_product_delete AFTER DELETE ON Products
FOR EACH ROW
BEGIN
 INSERT INTO price_audit(product_id,action_type,old_price,new_price)
 VALUES(OLD.product_id,'DELETE',OLD.unit_price,NULL);
END $$
DELIMITER ;
